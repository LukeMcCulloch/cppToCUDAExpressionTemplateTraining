7	template <typename Derived>
8	struct VecExpr 
9	{
10	    const Derived& self() const { return static_cast<const Derived&>(*this); } // get through to the real type.  -> cast this base to the derived type, so we can call derived methods from the base class
11	    
12	    double operator[](std::size_t i) const { return self()[i]; }
13	    std::size_t size() const { return self().size(); } //
14	};
15	
16	class Vector : public VecExpr<Vector>
17	{
18	public:
19	    explicit Vector(std::size_t n) : size_(n), data_(std::make_unique<double[]>(n)) {} // memsets to zero under the hood (or loops, but anyway), so no need to loop and zero-initialize ourselves
20	
21	
22	    
23	    Vector(const Vector& other) : size_(other.size_), data_(new double[other.size_]) {
24	        for (std::size_t i = 0; i < size_; ++i) data_[i] = other.data_[i];
25	    }
26	
27	
28	    Vector& operator=(const Vector& other) {
29	        if (this == &other) return *this;
30	        size_ = other.size_;
31	        data_.reset(new double[size_]);// raw new[], not make_unique, because we want to copy the elements over ourselves, not default-construct them
32	        for (std::size_t i = 0; i < size_; ++i) data_[i] = other.data_[i];
33	        return *this;
34	    }
35	
36	    double& operator[](std::size_t i) { return data_[i]; } 
37	    const double& operator[](std::size_t i) const { return data_[i]; }
38	    std::size_t size() const { return size_; }
39	
40	private:
41	    std::size_t size_;
42	    std::unique_ptr<double[]> data_;
43	};

46	template <typename T>
47	struct ExprTraits { using ExprRef = T; };
48	
49	
50	template <>
51	struct ExprTraits<Vector> { using ExprRef = const Vector&; };

53	template <typename LHS, typename RHS>
54	class AddExpr : public VecExpr<AddExpr<LHS, RHS>> {
55	public:
56	    AddExpr(const LHS& l, const RHS& r) : lhs_(l), rhs_(r) {}
57	
58	    double operator[](std::size_t i) const { return lhs_[i] + rhs_[i]; }// happily recurses through chains of additions.
59	    std::size_t size() const { return lhs_.size(); }
60	
61	private:
62	    typename ExprTraits<LHS>::ExprRef lhs_;
63	    typename ExprTraits<RHS>::ExprRef rhs_;
64	};
65	
66	template <typename LHS, typename RHS>
67	AddExpr<LHS, RHS> operator+(const VecExpr<LHS>& lhs, const VecExpr<RHS>& rhs) {
68	    return AddExpr<LHS, RHS>(lhs.self(), rhs.self());
69	}
70	
71	template <typename Derived>
72	Vector evaluate(const VecExpr<Derived>& expr) {
73	    Vector result(expr.size());// allocate a real Vector called "result" to hold the result
74	    for (std::size_t i = 0; i < expr.size(); ++i) result[i] = expr[i]; // very lazy evaluation: 
75	    return result;
76	}
